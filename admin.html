Google Sheets / Webhook Sync (For Live Domain Launch)

Once your custom domain is live, visitors submit quotes from their own phones. Paste your free Google Apps Script /exec URL below AND into crm-config.js so every customer quote from any phone syncs straight to your Google Sheet and this Admin Panel.

View 1-Click Free Google Sheets Apps Script Setup & Domain Checklist

Setup: Open Google Sheets → Extensions → Apps Script → paste the code below → Deploy as Web App (Execute as: Me, Who has access: Anyone) → copy the /exec URL into crm-config.js and above.

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var payload = JSON.parse(e.postData.contents);
  var lead = payload.lead || payload;
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID","Created","Name","Phone","Email","City","World","Service","SubService","Concept","Message","Status"]);
  }
  sheet.appendRow([lead.id, lead.createdAt, lead.name, lead.contact, lead.email, lead.city, lead.world, lead.service, lead.subService, lead.concept, lead.message, lead.status]);
  return ContentService.createTextOutput(JSON.stringify({ ok: true })).setMimeType(ContentService.MimeType.JSON);
}

function doGet() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  if (rows.length < 2) return ContentService.createTextOutput(JSON.stringify({ leads: [] })).setMimeType(ContentService.MimeType.JSON);
  var leads = rows.slice(1).map(function(r) {
    return { id: String(r[0]), createdAt: String(r[1]), name: String(r[2]), contact: String(r[3]), email: String(r[4]), city: String(r[5]), world: String(r[6]), service: String(r[7]), subService: String(r[8]), concept: String(r[9]), message: String(r[10]), status: String(r[11] || "New"), priority: "Warm", estimatedValue: 0, notes: [] };
  });
  return ContentService.createTextOutput(JSON.stringify({ leads: leads })).setMimeType(ContentService.MimeType.JSON);
}
